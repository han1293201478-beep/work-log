function generateChart(records) { const ctx = document.getElementById('financeChart').getContext('2d'); if (myChart) myChart.destroy(); // 如果没有数据，显示提示 if (!records || records.length === 0) { ctx.font = "14px sans-serif"; ctx.fillStyle = "#666"; ctx.textAlign = "center"; ctx.fillText("当前时段无数据", ctx.canvas.width/2, ctx.canvas.height/2); return; } if (chartMode === 'pie') { // --- 饼图：支出占比分析 --- const stats = {}; records.filter(r => r.amount < 0).forEach(r => { const tag = r.category_tag || '未分类'; stats[tag] = (stats[tag] || 0) + Math.abs(r.amount); }); // 性能优化：按金额大小排序，让图表更直观 const sortedLabels = Object.keys(stats).sort((a, b) => stats[b] - stats[a]); const dataValues = sortedLabels.map(label => stats[label]); // 在标签上直接显示总额 const displayLabels = sortedLabels.map(label => `${label}: ￥${Math.round(stats[label]).toLocaleString()}`); myChart = new Chart(ctx, { type: 'doughnut', data: { labels: displayLabels, datasets: [{ data: dataValues, backgroundColor: ['#8ab4f8', '#ff7875', '#95de64', '#ffc069', '#b37feb', '#5cdbd3'], borderWidth: 0 }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'right', labels: { color: '#a0a0a0', font: { size: 11 }, boxWidth: 12 } } } } }); } else { // --- 折线图：时间轴趋势 --- const timeline = {}; // 按照日期从小到大排序（正序） const sortedRecords = [...records].sort((a, b) => new Date(a.record_date) - new Date(b.record_date)); let runningTotal = 0; sortedRecords.forEach(r => { const date = r.record_date.split('T')[0]; runningTotal += r.amount; timeline[date] = runningTotal; }); myChart = new Chart(ctx, { type: 'line', data: { labels: Object.keys(timeline), datasets: [{ label: '余额走势', data: Object.values(timeline), borderColor: '#8ab4f8', backgroundColor: 'rgba(138, 180, 248, 0.1)', fill: true, tension: 0.3, pointRadius: 3, pointHitRadius: 10 }] }, options: { responsive: true, maintainAspectRatio: false, scales: { x: { ticks: { color: '#666', font: { size: 10 } }, grid: { display: false } }, y: { ticks: { color: '#666', font: { size: 10 } }, grid: { color: '#333' } } } } }); } }
